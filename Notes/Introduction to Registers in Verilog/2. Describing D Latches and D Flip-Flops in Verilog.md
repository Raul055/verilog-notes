# D Latches
The following module corresponds to a `D Latch`.
```verilog
module D_latch
	(
		input D,
		input clk,
		output reg Q,
		output Q_b
	);

	// D-Latch
	assign Q_b = ~Q;
	always @(D, clk)
	begin
		Q = Q;
		if (clk)
			Q = D;
		else
			Q = Q;
	end

endmodule
```

# D Flip-Flop
The following module corresponds to a `D Flip-Flop` activated by a `posedge` of the `clk`.
```verilog
module D_FF_pos
	(
		input D,
		input clk,
		output reg Q
	);
	
	// D Flip-Flop with posedge
	always @(posedge clk)
	begin
		Q = D;
	end

endmodule
```

Likewise, the following module corresponds to a `D Flip-Flop` activated by a `negedge` of the `clk`.

```verilog
module D_FF_neg
	(
		input D,
		input clk,
		output reg Q
	);

	// D Flip-Flop with negedge
	always @(negedge clk)
	begin
		Q = D;
	end

endmodule
```

# Test Circuit

The following corresponds to a test circuit which encapsulates the previous modules.
```verilog
module storage_elements_top
	(
		input D,
		input clk,
		output Qa,
		output Qb,
		output Qc
	);
	
	D_latch L0 (
		.D(D),
		.clk(clk),
		.Q(Qa),
		.Q_b()
	);
	
	D_FF_neg FF0 (
		.D(D),
		.clk(clk),
		.Q(Qb)
	);
	
	D_FF_pos FF1 (
		.D(D),
		.clk(clk),
		.Q(Qc)
	);

endmodule
```

And with it, we can create a testbench for out top module.

```verilog
// Timescale
`timescale 1ns/1ps

// Module
module storage_elements_tb;
	// Declare signals to connect to DUT
	// reg ---> inputs
	// wire --> outputs
	reg clk;
	reg D;
	wire Q_latch;
	wire Q_ff_pos;
	wire Q_ff_neg;
	
	// Instantiate uut
	storage_elements_top uut(
		.D(D),
		.clk(clk),
		.Qa(Q_latch),
		.Qb(Q_ff_pos),
		.Qc(Q_ff_neg)
	);
	
	// Generate stimuli
	localparam T = 20; // clk period
	always
	begin
		clk = 1'b0;
		#(T/2);
		clk = 1'b1;
		#(T/2);
	end
	
	// Begin test
	initial begin
		D = 1'b1;
		#(2 * T);
		D = 1'b0;
		@(posedge clk);
		D = 1'b1;
		#2 D = 1'b0;
		#3 D = 1'b1;
		#4 D = 1'b0;
		repeat(2) @(negedge clk);
		D = 1'b1;
		#20;
		$finish;
	end
	
	// Prints results to console
	initial begin
		$monitor("time=%0t D=%b Q_latch=%b Q_ff_pos=%b Q_ff_neg", $time,
		D, Q_latch, Q_ff_pos, Q_ff_neg);
	end
	
	// Dumps .vcd file
	initial begin
		`ifdef VCD_PATH
		$dumpfile(`VCD_PATH);
		`else
		$dumpfile("default.vcd");
		`endif
		$dumpvars(0, storage_elements_tb);
	end

endmodule
```

# Testbench Timing Control Structures
Some timing control structures for tesbenches with sequential circuits.
```verilog
#20;                        // Wait for 20 time units
#(2 * T)                    // Wait for 2T time units (T is a localparam)
@(negedge rst);             // Wait rst to deassert
@(negedge clk);             // Wait for the next negative clk edge
repeat(10) @(negedge clk);  // wait for 10 negative clk edges
wait(x == 2);               // Wait for x to become 2
wait(y);                    // Wait for y to become 1
$stop                       // Stop simulation
```
