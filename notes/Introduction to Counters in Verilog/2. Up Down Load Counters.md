# `N-Bits` Up Counter
The following module describe a `N-Bits` **Up Counter**.

```verilog
module up_counter
	#(parameter N=4)
	(
		input clk,
		input reset,
		output [N-1:0] Q
	);
	
	// Register
	reg [N-1:0] Q_reg, Q_next;
	always @(posedge clk, negedge reset)
	begin
		if(reset)
			Q_reg <= 0;
		else
			Q_reg <= Q_next;
	end
	
	// Next state logic
	always @(Q_reg)
	begin
		Q_next = Q_reg + 1;
	end
	
	// Ouput logic
	assign Q = Q_reg;
	
endmodule
```

This module can be checked with the following testbench.

```verilog
// Timescale
`timescale 1ns/1ps

// Module
module up_counter_tb;
	localparam N=4;
	reg clk;
	reg reset;
	wire [N-1:0] Q;
	
	// Instantiate uut
	up_counter #(.N(N)) uut(
		.clk(clk),
		.reset(reset),
		.Q(Q)
	);
	
	// Timer
	initial
		#200 $stop;
	
	// Generate stimuli
	
	// Generate clk signal
	localparam T = 10; // clk period
	always
	begin
		clk = 1'b0;
		#(T/2);
		clk = 1'b1;
		#(T/2);
	end
	
	// Begin test
	initial begin
		// Issue a quick reset for 2 ns
		reset = 1'b0;
		#2 reset = 1'b1;
		$finish;
	end
	
	// Prints results to console
	initial begin
		$monitor("time=%0t reset=%b Q=%b", $time,
		reset, Q);
	end
	
	// Dumps .vcd file
	initial begin
		`ifdef VCD_PATH
		$dumpfile(`VCD_PATH);
		`else
		$dumpfile("default.vcd");
		`endif
		$dumpvars(0, up_counter_tb);
	end

endmodule
```

# `N-Bits` Down Counter
The following module describe a `N-Bits` **Down Counter**. The same as the `N-Bits` Up Counter, but only changes in the *Next State Logic*.

```verilog
module down_counter
	#(parameter N=4)
	(
		input clk,
		input reset,
		output [N-1:0] Q
	);
	
	// Register
	reg [N-1:0] Q_reg, Q_next;
	always @(posedge clk, negedge reset)
	begin
		if(reset)
			Q_reg <= 0;
		else
			Q_reg <= Q_next;
	end
	
	// Next state logic
	always @(Q_reg)
	begin
		Q_next = Q_reg - 1;
	end
	
	// Ouput logic
	assign Q = Q_reg;
	
endmodule
```

# `N-Bits` Up/Down Counter

The following module represents a `N-Bits` **Up/Down Counter**.
```verilog
module up_down_counter
	#(parameter N=4)
	(
		input clk,
		input reset,
		input enable,
		input up, // when asserted the counter is up, else down counter
		output [N-1:0] Q
	);
	
	// Register
	reg [N-1:0] Q_reg, Q_next;
	always @(posedge clk, negedge reset)
	begin
		// Asynchronous reset
		if(reset)
			Q_reg <= 0;
		// Enable UP/DOWN counter
		else if(enable)
			Q_reg <= Q_next;
		// Counter disabled, maintain current value
		else
			Q_reg <= Q_reg;
	end
	
	// Next state logic
	always @(Q_reg, up)
	begin
		Q_next = Q_reg;
		if (up)
			Q_next = Q_reg + 1;
		else
			Q_next = Q_reg - 1;
	end
	
	// Ouput logic
	assign Q = Q_reg;
	
endmodule
```

This module can be checked with the following testbench.

# `N-Bits` Up/Down Load Counter
The following module represents a `N-Bits` **Up/Down Load Counter**.
```verilog
module up_down_load_counter
	#(parameter N=4)
	(
		input clk,
		input reset,
		input enable,
		input up, // when asserted the counter is up, else down counter
		input load,
		input [N-1:0] D,
		output [N-1:0] Q
	);
	
	// Register
	reg [N-1:0] Q_reg, Q_next;
	always @(posedge clk, negedge reset)
	begin
		// Asynchronous reset
		if(reset)
			Q_reg <= 0;
		// Enable UP/DOWN counter
		else if(enable)
			Q_reg <= Q_next;
		// Counter disabled, maintain current value
		else
			Q_reg <= Q_reg;
	end
	
	// Next state logic
	always @(*)
	begin
		Q_next = Q_reg;
		casex({load, up})
			2'b00: Q_next = Q_reg - 1;
			2'b01: Q_next = Q_reg + 1;
			2'b1x: Q_next = D;
			default: Q_next = Q_reg;
		endcase
	end
	
	// Ouput logic
	assign Q = Q_reg;
	
endmodule
```
